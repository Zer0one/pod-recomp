# Native development host: reuse the toolkit's HLE, not the x86 native bridge.
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
set(PCRECOMP "${PROJECT_SOURCE_DIR}/work/pcrecomp" CACHE PATH "Pinned pcrecomp checkout")
set(POD_SDL_TTF_SOURCE "${PROJECT_SOURCE_DIR}/work/SDL_ttf" CACHE PATH "Optional SDL_ttf source checkout")
pkg_check_modules(SDL_TTF QUIET IMPORTED_TARGET SDL2_ttf)
if(SDL_TTF_FOUND)
  set(POD_TTF_TARGET PkgConfig::SDL_TTF)
else()
  # Use the same SDL selected for the host. An unrelated system framework
  # discovered by SDL_ttf can otherwise load a second SDL2 implementation.
  if(NOT TARGET SDL2::SDL2)
    add_library(SDL2::SDL2 ALIAS PkgConfig::SDL2)
  endif()
  set(SDL2TTF_SAMPLES OFF CACHE BOOL "" FORCE)
  set(SDL2TTF_INSTALL OFF CACHE BOOL "" FORCE)
  set(SDL2TTF_VENDORED OFF CACHE BOOL "" FORCE)
  set(SDL2TTF_HARFBUZZ OFF CACHE BOOL "" FORCE)
  add_subdirectory("${POD_SDL_TTF_SOURCE}" "${CMAKE_BINARY_DIR}/sdl-ttf" EXCLUDE_FROM_ALL)
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/ttf-include/SDL2")
  configure_file("${POD_SDL_TTF_SOURCE}/SDL_ttf.h"
    "${CMAKE_BINARY_DIR}/ttf-include/SDL2/SDL_ttf.h" COPYONLY)
  set(POD_TTF_TARGET SDL2_ttf)
endif()

set(STAGED "${CMAKE_BINARY_DIR}/pcrecomp-runtime")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${PROJECT_SOURCE_DIR}/tools/macos/prepare_runtime.py"
  "${PROJECT_SOURCE_DIR}/tools/macos/patches/win32hle.patch")
execute_process(COMMAND "${Python3_EXECUTABLE}"
  "${PROJECT_SOURCE_DIR}/tools/macos/prepare_runtime.py" "${PCRECOMP}" "${STAGED}"
  COMMAND_ERROR_IS_FATAL ANY)
set(HLE "${STAGED}/runtime/win32hle")
file(GLOB HLE_SOURCES "${HLE}/*.c")
list(FILTER HLE_SOURCES EXCLUDE REGEX "_selftest\\.c$")
add_library(pod_hle STATIC ${HLE_SOURCES})
target_include_directories(pod_hle PUBLIC "${HLE}" "${STAGED}/runtime/recomp32")
target_include_directories(pod_hle PRIVATE "${CMAKE_BINARY_DIR}/ttf-include")
target_link_libraries(pod_hle PUBLIC PkgConfig::SDL2 ${POD_TTF_TARGET} Threads::Threads)
target_compile_definitions(pod_hle PUBLIC RECOMP_RETADDR=0u RECOMP_CPUID_PENTIUM_MMX)
target_compile_options(pod_hle PRIVATE -fno-strict-aliasing -Wno-unused-function)

set(POD_GUEST_VARIANT "mmx" CACHE STRING "Guest build: mmx, or an isolated renderer probe")
set_property(CACHE POD_GUEST_VARIANT PROPERTY STRINGS mmx glide d3d)
set(POD_GUEST_DESKTOP_BPP 0)
if(POD_GUEST_VARIANT STREQUAL "mmx")
  set(POD_GUEST_IMAGE "PODMMX.EXE")
  set(POD_GEN_DIR "${PROJECT_SOURCE_DIR}/src/recomp/gen")
elseif(POD_GUEST_VARIANT STREQUAL "glide" OR POD_GUEST_VARIANT STREQUAL "d3d")
  if(POD_GUEST_VARIANT STREQUAL "glide")
    set(POD_GUEST_IMAGE "PODX3DFX.EXE")
    set(POD_GUEST_DESKTOP_BPP 16)
  else()
    set(POD_GUEST_IMAGE "PODD3DX.EXE")
  endif()
  set(POD_GEN_DIR "${PROJECT_SOURCE_DIR}/work/renderers/${POD_GUEST_VARIANT}/gen")
  message(STATUS "Alternate guest probe: experimental renderer, fidelity unqualified")
else()
  message(FATAL_ERROR "Unknown POD_GUEST_VARIANT: ${POD_GUEST_VARIANT}")
endif()
file(GLOB GEN CONFIGURE_DEPENDS "${POD_GEN_DIR}/*.c")
if(NOT GEN)
  if(POD_GUEST_VARIANT STREQUAL "mmx")
    message(FATAL_ERROR "Run tools/catalog.py and run_lift.py with the pinned toolkit first")
  else()
    message(FATAL_ERROR "Run tools/portable/renderer_probe.py ${POD_GUEST_VARIANT} --stage lift first")
  endif()
endif()
set(POD_GEN_OPT "-O1" CACHE STRING "Optimization for translated guest code")
add_library(pod_guest OBJECT ${GEN})
target_include_directories(pod_guest PRIVATE "${POD_GEN_DIR}")
target_link_libraries(pod_guest PRIVATE pod_hle)
target_compile_options(pod_guest PRIVATE -w -fno-strict-aliasing ${POD_GEN_OPT})
target_compile_definitions(pod_guest PRIVATE RECOMP_LOCAL_REGS)
# Services have no main(), command-line parser or Cocoa lifecycle ownership.
# PCM output is linked by the frontend; codec/HLE still depend on SDL.
add_library(pod_services STATIC dinput.c audio.c codec.c cd_audio.c compat.c video_present.c)
target_link_libraries(pod_services PUBLIC pod_hle)
target_compile_options(pod_services PRIVATE -Wall -Wextra -fno-strict-aliasing)
add_library(pod_pcm_sdl STATIC pcm_sdl.c)
target_link_libraries(pod_pcm_sdl PUBLIC PkgConfig::SDL2)
add_executable(pod host.c input_sdl.c $<TARGET_OBJECTS:pod_guest>)
target_link_libraries(pod PRIVATE pod_services pod_pcm_sdl)
target_compile_options(pod PRIVATE -Wall -Wextra -fno-strict-aliasing)
target_compile_definitions(pod PRIVATE POD_SOURCE_DIR="${PROJECT_SOURCE_DIR}" POD_GUEST_IMAGE="${POD_GUEST_IMAGE}"
  POD_GUEST_DESKTOP_BPP=${POD_GUEST_DESKTOP_BPP})
set_target_properties(pod PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
if(POD_GUEST_VARIANT STREQUAL "glide")
  if(NOT APPLE)
    message(FATAL_ERROR "The experimental Glide GPU backend currently requires the macOS backend")
  endif()
  enable_language(OBJC)
  set(POD_GLIDE_SHADER "${CMAKE_CURRENT_SOURCE_DIR}/../macos/glide_shader.metal")
  set(POD_GLIDE_LAYOUT "${CMAKE_CURRENT_SOURCE_DIR}/glide_state_layout.h")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${POD_GLIDE_SHADER}" "${POD_GLIDE_LAYOUT}")
  file(READ "${POD_GLIDE_LAYOUT}" POD_LAYOUT_TEXT)
  file(READ "${POD_GLIDE_SHADER}" POD_SHADER_TEXT)
  string(PREPEND POD_SHADER_TEXT "${POD_LAYOUT_TEXT}\n")
  string(REPLACE "\\" "\\\\" POD_SHADER_TEXT "${POD_SHADER_TEXT}")
  string(REPLACE "\"" "\\\"" POD_SHADER_TEXT "${POD_SHADER_TEXT}")
  string(REPLACE "\n" "\\n\"\n\"" POD_SHADER_TEXT "${POD_SHADER_TEXT}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/glide_shaders.h" "static const char *pod_glide_shader =\n\"${POD_SHADER_TEXT}\";\n")
  add_library(pod_glide_gpu STATIC ../macos/glide_metal.m)
  target_include_directories(pod_glide_gpu PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_include_directories(pod_glide_gpu PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_options(pod_glide_gpu PRIVATE -Wall -Wextra -fno-objc-arc)
  target_link_libraries(pod_glide_gpu PUBLIC "-framework Metal" "-framework Foundation")
  add_library(pod_glide_bootstrap STATIC glide_bootstrap.c glide_vertex.c glide_texture.c)
  target_link_libraries(pod_glide_bootstrap PUBLIC pod_hle pod_glide_gpu)
  add_library(pod_glide_compat STATIC glide_compat.c)
  target_link_libraries(pod_glide_compat PUBLIC pod_hle)
  target_compile_options(pod_glide_compat PRIVATE -Wall -Wextra -fno-strict-aliasing)
  target_link_libraries(pod PRIVATE pod_glide_bootstrap pod_glide_compat)
  target_compile_definitions(pod PRIVATE POD_GLIDE_BOOTSTRAP)
  add_executable(pod_glide_lifecycle_selftest glide_lifecycle_selftest.c)
  target_link_libraries(pod_glide_lifecycle_selftest PRIVATE pod_glide_compat)
  add_test(NAME pod_glide_lifecycle COMMAND pod_glide_lifecycle_selftest)
  set_tests_properties(pod_glide_lifecycle PROPERTIES TIMEOUT 10
    ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
  add_executable(pod_glide_gpu_selftest glide_gpu_selftest.c glide_vertex.c glide_texture.c)
  target_link_libraries(pod_glide_gpu_selftest PRIVATE pod_glide_gpu)
  add_test(NAME pod_glide_gpu COMMAND pod_glide_gpu_selftest)
  set_tests_properties(pod_glide_gpu PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 20)
  add_executable(pod_glide_service_selftest glide_service_selftest.c)
  target_link_libraries(pod_glide_service_selftest PRIVATE pod_glide_bootstrap)
  add_test(NAME pod_glide_service COMMAND pod_glide_service_selftest)
  set_tests_properties(pod_glide_service PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 20
    ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
endif()

  set_target_properties(pod PROPERTIES OUTPUT_NAME "pod-${POD_GUEST_VARIANT}-probe")
