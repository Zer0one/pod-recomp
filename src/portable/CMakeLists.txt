# Native development host: reuse the toolkit's HLE, not the x86 native bridge.
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
set(PCRECOMP "${PROJECT_SOURCE_DIR}/work/pcrecomp" CACHE PATH "Pinned pcrecomp checkout")
set(POD_SDL_TTF_SOURCE "${PROJECT_SOURCE_DIR}/work/SDL_ttf" CACHE PATH "Optional SDL_ttf source checkout")
pkg_check_modules(SDL_TTF QUIET IMPORTED_TARGET SDL2_ttf)
if(SDL_TTF_FOUND)
  set(POD_TTF_TARGET PkgConfig::SDL_TTF)
else()
  # Use the same SDL selected for the host. An unrelated system framework
  # discovered by SDL_ttf can otherwise load a second SDL2 implementation.
  if(NOT TARGET SDL2::SDL2)
    add_library(SDL2::SDL2 ALIAS PkgConfig::SDL2)
  endif()
  set(SDL2TTF_SAMPLES OFF CACHE BOOL "" FORCE)
  set(SDL2TTF_INSTALL OFF CACHE BOOL "" FORCE)
  set(SDL2TTF_VENDORED OFF CACHE BOOL "" FORCE)
  set(SDL2TTF_HARFBUZZ OFF CACHE BOOL "" FORCE)
  add_subdirectory("${POD_SDL_TTF_SOURCE}" "${CMAKE_BINARY_DIR}/sdl-ttf" EXCLUDE_FROM_ALL)
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/ttf-include/SDL2")
  configure_file("${POD_SDL_TTF_SOURCE}/SDL_ttf.h"
    "${CMAKE_BINARY_DIR}/ttf-include/SDL2/SDL_ttf.h" COPYONLY)
  set(POD_TTF_TARGET SDL2_ttf)
endif()

set(STAGED "${CMAKE_BINARY_DIR}/pcrecomp-runtime")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${PROJECT_SOURCE_DIR}/tools/macos/prepare_runtime.py"
  "${PROJECT_SOURCE_DIR}/tools/macos/patches/win32hle.patch")
execute_process(COMMAND "${Python3_EXECUTABLE}"
  "${PROJECT_SOURCE_DIR}/tools/macos/prepare_runtime.py" "${PCRECOMP}" "${STAGED}"
  COMMAND_ERROR_IS_FATAL ANY)
set(HLE "${STAGED}/runtime/win32hle")
file(GLOB HLE_SOURCES "${HLE}/*.c")
list(FILTER HLE_SOURCES EXCLUDE REGEX "_selftest\\.c$")
add_library(pod_hle STATIC ${HLE_SOURCES})
target_include_directories(pod_hle PUBLIC "${HLE}" "${STAGED}/runtime/recomp32")
target_include_directories(pod_hle PRIVATE "${CMAKE_BINARY_DIR}/ttf-include")
target_link_libraries(pod_hle PUBLIC PkgConfig::SDL2 ${POD_TTF_TARGET} Threads::Threads)
target_compile_definitions(pod_hle PUBLIC RECOMP_RETADDR=0u RECOMP_CPUID_PENTIUM_MMX)
target_compile_options(pod_hle PRIVATE -fno-strict-aliasing -Wno-unused-function)

file(GLOB GEN CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/src/recomp/gen/*.c")
if(NOT GEN)
  message(FATAL_ERROR "Run tools/catalog.py and run_lift.py with the pinned toolkit first")
endif()
set(POD_GEN_OPT "-O1" CACHE STRING "Optimization for translated guest code")
add_library(pod_guest OBJECT ${GEN})
target_include_directories(pod_guest PRIVATE "${PROJECT_SOURCE_DIR}/src/recomp/gen")
target_link_libraries(pod_guest PRIVATE pod_hle)
target_compile_options(pod_guest PRIVATE -w -fno-strict-aliasing ${POD_GEN_OPT})
target_compile_definitions(pod_guest PRIVATE RECOMP_LOCAL_REGS)
# Services have no main(), command-line parser or Cocoa lifecycle ownership.
# PCM output is linked by the frontend; codec/HLE still depend on SDL.
add_library(pod_services STATIC dinput.c audio.c codec.c cd_audio.c compat.c video_present.c)
target_link_libraries(pod_services PUBLIC pod_hle)
target_compile_options(pod_services PRIVATE -Wall -Wextra -fno-strict-aliasing)
add_library(pod_pcm_sdl STATIC pcm_sdl.c)
target_link_libraries(pod_pcm_sdl PUBLIC PkgConfig::SDL2)
add_executable(pod host.c input_sdl.c $<TARGET_OBJECTS:pod_guest>)
target_link_libraries(pod PRIVATE pod_services pod_pcm_sdl)
target_compile_options(pod PRIVATE -Wall -Wextra -fno-strict-aliasing)
target_compile_definitions(pod PRIVATE POD_SOURCE_DIR="${PROJECT_SOURCE_DIR}")
set_target_properties(pod PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
if(APPLE)
  set(POD_APP "${CMAKE_BINARY_DIR}/POD.app")
  file(MAKE_DIRECTORY "${POD_APP}/Contents/MacOS")
  configure_file(Info.plist "${POD_APP}/Contents/Info.plist" COPYONLY)
  add_custom_command(TARGET pod POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
    "$<TARGET_FILE:pod>" "${POD_APP}/Contents/MacOS/pod")
endif()
add_executable(pod_portable_selftest selftest.c dinput.c input_sdl.c audio.c cd_audio.c)
target_link_libraries(pod_portable_selftest PRIVATE pod_hle pod_pcm_sdl)
add_test(NAME pod_portable COMMAND pod_portable_selftest)
add_executable(pod_cd_selftest cd_audio_selftest.c cd_audio.c)
target_link_libraries(pod_cd_selftest PRIVATE pod_hle pod_pcm_sdl)
add_test(NAME pod_cd COMMAND pod_cd_selftest)
# Same waveOut service, independently linked against a manually consumed queue.
add_executable(pod_audio_queue_selftest audio_queue_selftest.c audio.c)
target_link_libraries(pod_audio_queue_selftest PRIVATE pod_hle)
add_test(NAME pod_audio_queue COMMAND pod_audio_queue_selftest)
add_executable(pod_messages_selftest messages_selftest.c)
target_link_libraries(pod_messages_selftest PRIVATE pod_hle)
add_test(NAME pod_messages COMMAND pod_messages_selftest)
add_executable(pod_send_callback_selftest send_callback_selftest.c)
target_link_libraries(pod_send_callback_selftest PRIVATE pod_hle)
add_test(NAME pod_send_callback COMMAND pod_send_callback_selftest)
foreach(name pe_loader host pump)
  add_executable(hle_${name}_selftest "${HLE}/${name}_selftest.c")
  target_link_libraries(hle_${name}_selftest PRIVATE pod_hle)
  add_test(NAME hle_${name} COMMAND hle_${name}_selftest)
endforeach()
set_tests_properties(pod_portable pod_cd pod_audio_queue pod_messages pod_send_callback hle_pe_loader hle_host hle_pump PROPERTIES
  ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy" TIMEOUT 30)
